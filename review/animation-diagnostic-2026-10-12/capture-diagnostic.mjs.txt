import { chromium } from '/home/ali/Desktop/zawedni-story-3d/node_modules/playwright-core/index.mjs';
import { mkdirSync, writeFileSync, renameSync, readFileSync } from 'node:fs';
import assert from 'node:assert/strict';

const root = '/home/ali/Desktop/zawedni-story-3d';
const out = `${root}/review/animation-diagnostic-2026-10-12`;
mkdirSync(`${out}/raw`, { recursive: true });
const report = { date: new Date().toISOString(), sourceUnmodified: true, base: 'http://127.0.0.1:4318',
  launcher: 'ZS3D_NO_BROWSER=1 XDG_RUNTIME_DIR=/tmp ./present.sh', normalSpeed: true,
  audioRecorded: false, cases: [], additional: [], errors: [], external: [], failedRequests: [], clips: [] };
const args = process.env.DIAGNOSTIC_GL === '1'
  ? ['--use-angle=gl', '--use-gl=angle', '--ignore-gpu-blocklist']
  : ['--use-angle=swiftshader', '--enable-unsafe-swiftshader'];
const browser = await chromium.launch({ channel: 'chrome', headless: process.env.DIAGNOSTIC_GL !== '1',
  args: [...args, '--disable-background-timer-throttling', '--disable-renderer-backgrounding', '--disable-backgrounding-occluded-windows'] });
const context = await browser.newContext({ viewport: { width: 960, height: 540 }, reducedMotion: 'no-preference',
  recordVideo: { dir: `${out}/raw`, size: { width: 960, height: 540 } } });
context.on('request', r => { if (/^https?:/.test(r.url()) && !r.url().startsWith(report.base)) report.external.push(r.url()); });
context.on('response', r => { if (r.status() >= 400) report.failedRequests.push({ url: r.url(), status: r.status() }); });
const jury = await context.newPage();
jury.on('pageerror', e => report.errors.push(e.message));
const video = jury.video();
let notes;
try {
  const popup = context.waitForEvent('page');
  await jury.goto(`${report.base}/?pv=auto#probleme-0`, { waitUntil: 'domcontentloaded' });
  notes = await popup;
  notes.on('pageerror', e => report.errors.push(e.message));
  await jury.waitForFunction(() => window.__story?.ready, null, { timeout: 90000 });
  await notes.waitForFunction(() => window.__presenter?.cur === 'probleme-0', null, { timeout: 20000 });
  await jury.bringToFront();
  await jury.waitForTimeout(1800);
  report.build = await (await jury.request.get(`${report.base}/build-info.json`)).json().then(x => ({ buildId: x.buildId, projectRoot: x.projectRoot }));
  report.environment = await jury.evaluate(() => {
    const gl = __story.renderer.getContext(), ext = gl.getExtension('WEBGL_debug_renderer_info');
    return { renderer: ext ? gl.getParameter(ext.UNMASKED_RENDERER_WEBGL) : gl.getParameter(gl.RENDERER),
      reducedMotion: matchMedia('(prefers-reduced-motion: reduce)').matches, storedMotion: localStorage.getItem('zs3d-motion'),
      qualityLight: document.body.classList.contains('quality-light'), search: location.search };
  });
  await jury.evaluate(() => {
    window.__diagnostic = { active: false, samples: [], commands: [] };
    const d = __story.director, frame = d.frame.bind(d), go = d.go.bind(d);
    d.go = (i, opts) => {
      const before = { id: d.state.id, display: d.renderedState.id, opacity: d.contentOpacity,
        camera: d.camera.position.toArray(), guide: d.renderedPoses.guide?.pos.toArray(), time: performance.now() / 1000 };
      go(i, opts);
      if (__diagnostic.active) __diagnostic.commands.push({ before, to: d.state.id, t0: d.tr?.t0, timing: d.tr?.timing,
        from: d.tr?.from.id, pathLength: d.tr?.paths.guide?.length ?? 0,
        capturedCamera: d.tr?.shot.pos, capturedGuide: d.tr?.starts.guide?.pos.toArray(), initialOpacity: d.tr?.initialOpacity });
    };
    d.frame = now => {
      const tr = d.tr;
      frame(now);
      if (!__diagnostic.active) return;
      const stage = document.getElementById('stage'), evidence = document.getElementById('p-evidence');
      const p = d.renderedPoses.guide;
      __diagnostic.samples.push({ t: now, elapsed: tr ? now - tr.t0 : null, id: d.state.id, display: d.renderedState.id,
        phase: stage.dataset.phase, stageOpacity: +stage.style.opacity, stageVisibility: stage.style.visibility,
        evidenceOn: evidence.classList.contains('on'), evidenceFrame: evidence.dataset.frame,
        evidenceOpacity: +getComputedStyle(evidence).opacity, camera: d.camera.position.toArray(), target: d.camera.userData.lookTarget.toArray(),
        fov: d.camera.fov, guide: p.pos.toArray(), guideOpacity: p.opacity, walkW: p.walkW, walkDist: p.walkDist,
        activePanels: [...document.querySelectorAll('.pane.on')].map(e => e.id), timing: tr?.timing });
    };
  });
  async function at(id) {
    await jury.bringToFront();
    await jury.waitForFunction(id => __story.state === id && __story.director.renderedState.id === id && __story.settled && document.getElementById('stage').style.opacity === '1', id, { timeout: 25000 });
    await notes.waitForFunction(id => __presenter.cur === id, id, { timeout: 10000 });
  }
  async function prepare(id) {
    await jury.evaluate(id => {
      __diagnostic.active = false;
      const i = __story.states.findIndex(s => s.id === id);
      if (i < 0) throw new Error(`Unknown state ${id}`);
      __story.go(i, { instant: true });
    }, id);
    await at(id);
    await jury.waitForTimeout(600);
  }
  async function begin() {
    return await jury.evaluate(() => {
      __diagnostic.samples = []; __diagnostic.commands = []; __diagnostic.active = true;
      return performance.now() / 1000;
    });
  }
  async function end(label, from, to, start, clip = true) {
    await at(to);
    await jury.waitForTimeout(1300);
    const data = await jury.evaluate(() => ({ ...__diagnostic, end: performance.now() / 1000 }));
    await jury.evaluate(() => { __diagnostic.active = false; });
    const samples = data.samples;
    assert.ok(samples.length);
    assert.equal(samples.at(-1).phase, 'done');
    assert.equal(samples.at(-1).stageOpacity, 1);
    assert.equal(samples.at(-1).walkW, 0);
    for (const sample of samples) if (['move','settle'].includes(sample.phase)) {
      assert.equal(sample.stageOpacity, 0);
      assert.equal(sample.stageVisibility, 'hidden');
    }
    const summary = { label, from, to, start, end: data.end, commands: data.commands,
      phases: [...new Set(samples.map(s => s.phase))], frames: samples.length,
      minimumStageOpacity: Math.min(...samples.map(s => s.stageOpacity)),
      maxWalkWeight: Math.max(...samples.map(s => s.walkW)),
      cameraDisplacement: Math.max(...samples.map(s => Math.hypot(...s.camera.map((v,i) => v-samples[0].camera[i])))),
      guideDisplacement: Math.max(...samples.map(s => Math.hypot(...s.guide.map((v,i) => v-samples[0].guide[i])))) };
    report.cases.push(summary);
    writeFileSync(`${out}/${String(report.cases.length).padStart(2,'0')}-samples.json`, JSON.stringify(samples, null, 2));
    if (clip) report.clips.push({ label, from, to, start: start - .8, end: data.end });
    console.log(JSON.stringify(summary));
  }
  const cases = [
    ['Révélation de texte', 'probleme-0', 'probleme-1', 'ArrowRight'],
    ['Changement de capture', 'parcours-app-1', 'parcours-app-2', 'ArrowRight'],
    ['Nouvelle slide, caméra fixe', 'startup-1', 'probleme-0', 'ArrowRight'],
    ['Rapprochement vers les preuves de l’atelier', 'plateforme-2', 'sequence-0', 'ArrowRight'],
    ['Rapprochement vers le cycle 3D', 'devsecops-0', 'devsecops-1', 'ArrowRight'],
    ['Retour au monde', 'retour-0', 'retour-1', 'ArrowRight'],
    ['Déplacement entre chapitres', 'methodologie-1', 'ch2-0', 'ArrowRight'],
    ['Entrée dans le chapitre sans marche', 'ch2-0', 'outils-dev-0', 'ArrowRight'],
    ['Retour entre chapitres', 'ch2-0', 'methodologie-1', 'ArrowLeft'],
  ];
  for (const [label, from, to, key] of cases) {
    await prepare(from); const start = await begin(); await jury.keyboard.press(key); await end(label, from, to, start);
  }
  await prepare('methodologie-1');
  let start = await begin(); await jury.keyboard.press('ArrowRight');
  await jury.waitForFunction(() => document.getElementById('stage').dataset.phase === 'move', null, { timeout: 10000 });
  await jury.waitForTimeout(600); await jury.keyboard.press('ArrowLeft');
  await end('Interruption d’un déplacement par retour', 'methodologie-1', 'methodologie-1', start);
  await prepare('sequence-0');
  start = await begin();
  for (const key of ['ArrowRight','ArrowRight','ArrowRight','ArrowLeft']) {
    await jury.keyboard.press(key); await jury.waitForTimeout(65);
  }
  await end('Clics rapides à caméra fixe', 'sequence-0', 'sequence-2', start);
  // Separate from the video excerpts: measure the actual timing constructor on every
  // adjacent pair using current states and current anchors, without changing time.
  report.census = await jury.evaluate(() => {
    const d = __story.director, result = [];
    __diagnostic.active = false;
    for (let i=1;i<__story.states.length;i++) {
      d.go(i-1,{instant:true}); d.frame(performance.now()/1000);
      const from=d.state; d.go(i); const tr=d.tr;
      result.push({from:from.id,to:d.state.id,fromShot:from.shot,toShot:d.state.shot,
        sameShot:from.shot===d.state.shot,sameSource:from.source.seq===d.state.source.seq,
        sameGuide:from.actors.guide.at===d.state.actors.guide.at,
        sharedPanels:from.panels.filter(p=>d.state.panels.includes(p)), move:d.state.move,
        stateDuration:d.state.duration,timing:tr.timing,pathLength:tr.paths.guide?.length??0});
    }
    d.go(0,{instant:true}); return result;
  });
  // Quality and reduced motion affect different things; test both through real keys.
  await jury.keyboard.press('l');
  await prepare('probleme-0');start=await begin();await jury.keyboard.press('ArrowRight');
  await end('Texte en qualité légère (vérification hors montage)', 'probleme-0','probleme-1',start,false);
  await prepare('methodologie-1');start=await begin();await jury.keyboard.press('ArrowRight');
  await end('Voyage en qualité légère (vérification hors montage)', 'methodologie-1','ch2-0',start,false);
  await jury.keyboard.press('m');
  await prepare('methodologie-1');start=await begin();await jury.keyboard.press('ArrowRight');
  await end('Mouvement réduit (vérification hors montage)', 'methodologie-1','ch2-0',start,false);
  assert.equal(report.cases.at(-1).commands[0].timing.movement,0);
  assert.equal(report.errors.length,0);
  assert.equal(report.external.length,0);
  assert.equal(report.failedRequests.length,0);
} catch (e) {
  report.failure = e.stack;
  console.error(e);
} finally {
  await context.close(); await browser.close();
  renameSync(await video.path(), `${out}/jury-normal.webm`);
  writeFileSync(`${out}/report.json`,JSON.stringify(report,null,2)+'\n');
}
if(report.failure) process.exitCode=1;

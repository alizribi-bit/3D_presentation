from pathlib import Path
import json, subprocess
p = Path('/home/ali/Desktop/zawedni-story-3d/review/animation-diagnostic-2026-10-12')
normal = json.loads((p/'report.json').read_text())
rapid = json.loads((p/'rapid/report.json').read_text())
assert not normal.get('failure') and not rapid.get('failure')
clips = []
offset = 0
for source, cases in [(0,[normal['cases'][i] for i in [0,1,2,3,4,5,6,9]]),(1,rapid['cases'])]:
    for c in cases:
        first,last = c['commands'][0],c['commands'][-1]
        start = first['t0'] - .65
        end = min(c['end'], last['t0'] + last['timing']['total'] + 1)
        clip = dict(label=c['label'], fromState=c.get('from',first['before']['id']), toState=c.get('to',last['id'] if 'id' in last else last['to']), source=source, rawStart=start, rawEnd=end, start=offset, end=offset+end-start)
        clips.append(clip)
        offset += end - start
filters = ';'.join(f'[{c["source"]}:v]trim=start={c["rawStart"]}:end={c["rawEnd"]},setpts=PTS-STARTPTS[s{i}]' for i,c in enumerate(clips))+';'+''.join(f'[s{i}]' for i in range(len(clips)))+f'concat=n={len(clips)}:v=1:a=0[v]'
subprocess.run(['ffmpeg','-y','-v','error','-i',str(p/'jury-normal.webm'),'-i',str(p/'rapid/jury.webm'),'-filter_complex',filters,'-map','[v]','-c:v','libx264','-preset','fast','-crf','20','-pix_fmt','yuv420p','-movflags','+faststart',str(p/'diagnostic-normal.mp4')],check=True)
def stamp(t):
    ms=round(t*1000)
    return f'{ms//3600000:02}:{ms//60000%60:02}:{ms//1000%60:02}.{ms%1000:03}'
(p/'diagnostic-normal.vtt').write_text('WEBVTT\n\n'+'\n\n'.join(f'{stamp(c["start"])} --> {stamp(c["end"])}\n{c["label"]} : {c["fromState"]} → {c["toState"]}' for c in clips)+'\n')
(p/'recording.json').write_text(json.dumps(dict(normalSpeed=True,audioRecorded=False,timeManipulated=False,editing='Coupes uniquement, sans changement de vitesse ni interpolation des images; les deux dernières séquences utilisent les boutons DOM des notes.',renderer=normal['environment']['renderer'],clips=clips,duration=offset),ensure_ascii=False,indent=2)+'\n')
print('Saved normal-speed excerpt:',round(offset,2),'seconds')
